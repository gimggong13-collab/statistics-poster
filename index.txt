            document.getElementById('login-section').classList.add('hidden');
            document.getElementById('dashboard-section').classList.remove('hidden');
            document.getElementById('user-info-display').textContent = `${id} ${name}`;
            
            // 기존 데이터가 있다면 불러오기 시도 (오류 방지를 위해 지연 실행)
            setTimeout(loadStudentData, 100);
        }

        async function loadStudentData() {
            try {
                const { data, error } = await supabaseClient
                    .from('student_data')
                    .select('*')
                    .eq('id', currentUser.id)
                    .single();
                
                if (data) applyDataToUI(data);
            } catch (e) { console.log("불러올 기존 데이터가 없음"); }
        }

        function addAnalysisBox(data = null) {
            const container = document.getElementById('analysis-container');
            const div = document.createElement('div');
            div.className = "analysis-box animate-fade-in";
            div.innerHTML = `
                <button onclick="this.parentElement.remove()" class="absolute top-2 right-3 text-slate-300 hover:text-red-500 text-lg">&times;</button>
                <div class="grid grid-cols-1 md:grid-cols-3 gap-4">
                    <div class="md:col-span-1"><label class="text-[10px] font-black text-slate-400 uppercase tracking-tighter">문항 제목</label><input type="text" class="s3-item-title" placeholder="예: 운동 횟수" value="${data ? data.title : ''}"></div>

        function updateCharCount(el, targetId, min) {
            const len = el.value.length;
            const target = document.getElementById(targetId);
            target.textContent = `${len} / ${min}자`;
            target.className = len >= min ? 'text-right text-xs mt-2 text-green-600 font-bold' : 'text-right text-xs mt-2 text-slate-400 font-bold';
        }

        // 저장 로직 (Supabase + LocalStorage 백업)
        async function saveStep(step) {
            let stepData = {};
            if (step === 1) {
                stepData = {
                    analysis: document.getElementById('s1-graph-analysis').value,
                    q_pop: document.getElementById('s1-q-pop').value,
                    q_sample: document.getElementById('s1-q-sample').value,
                    career: document.getElementById('s1-career').value,
                    career_data: document.getElementById('s1-career-data').value,
                    questions: Array.from(document.querySelectorAll('.s1-q-item')).map(i => i.value)
                };
            } else if (step === 2) {
                stepData = {
                    method: document.getElementById('s2-method').value,
                    scope: document.getElementById('s2-scope').value,
                    link: document.getElementById('s2-link').value,
                    details: Array.from(document.querySelectorAll('.s2-question-detail')).map(i => i.value)
                };
            } else if (step === 3) {
                const text = document.getElementById('s3-interpretation').value;
                if(text.length < 150) { alert("해석 글자수가 부족합니다."); return; }
                const items = Array.from(document.querySelectorAll('.analysis-box')).map(box => ({
                    title: box.querySelector('.s3-item-title').value,
                    value: box.querySelector('.s3-item-value').value
                }));
                stepData = { items, canva: document.getElementById('s3-canva-link').value, tool: document.getElementById('s3-tool-link').value, text };
            } else if (step === 4) {
                const text = document.getElementById('s4-reflection').value;
                if(!document.getElementById('s4-check').checked) { alert("업로드 여부를 체크해주세요."); return; }
                if(text.length < 150) { alert("소감 글자수가 부족합니다."); return; }
                stepData = { final_link: document.getElementById('s4-final-link').value, reflection: text };
            }

            try {
                // Supabase 서버 저장 시도
                const { error } = await supabaseClient
                    .from('student_data')
                    .upsert({ 
                        id: currentUser.id, 
                        name: currentUser.name, 
                        [`s${step}`]: stepData, 
                        updated_at: new Date() 
                    });

                if (error) throw error;
                alert("성공적으로 저장되었습니다.");
                closeModal(step);
                if(step === 4) window.print();
            } catch (err) {
                console.error(err);
                localStorage.setItem(`stat_backup_${currentUser.id}_s${step}`, JSON.stringify(stepData));
                alert("서버 연결 불안정으로 브라우저에 임시 저장되었습니다.");
                closeModal(step);
            }
        }

        function applyDataToUI(data) {
            if (data.s1) {
                document.getElementById('s1-graph-analysis').value = data.s1.analysis || "";
                document.getElementById('s1-q-pop').value = data.s1.q_pop || "";
                document.getElementById('s1-q-sample').value = data.s1.q_sample || "";
                document.getElementById('s1-career').value = data.s1.career || "";
                document.getElementById('s1-career-data').value = data.s1.career_data || "";
                const qs = document.querySelectorAll('.s1-q-item');
                (data.s1.questions || []).forEach((v, idx) => { if(qs[idx]) qs[idx].value = v; });
            }
            if (data.s2) {
                document.getElementById('s2-method').value = data.s2.method || "";
                document.getElementById('s2-scope').value = data.s2.scope || "";
                document.getElementById('s2-link').value = data.s2.link || "";
                const ds = document.querySelectorAll('.s2-question-detail');
                (data.s2.details || []).forEach((v, idx) => { if(ds[idx]) ds[idx].value = v; });
            }
            if (data.s3) {
                document.getElementById('analysis-container').innerHTML = "";
                (data.s3.items || []).forEach(i => addAnalysisBox(i));
                document.getElementById('s3-canva-link').value = data.s3.canva || "";
                document.getElementById('s3-tool-link').value = data.s3.tool || "";
                document.getElementById('s3-interpretation').value = data.s3.text || "";
                updateCharCount(document.getElementById('s3-interpretation'), 's3-char-count', 150);
            }
            if (data.s4) {
                document.getElementById('s4-final-link').value = data.s4.final_link || "";
                document.getElementById('s4-reflection').value = data.s4.reflection || "";
                updateCharCount(document.getElementById('s4-reflection'), 's4-char-count', 150);
            }
        }

        async function checkAdminPassword() {
            const password = document.getElementById('admin-password-input').value;
            try {
                const { data, error } = await supabaseClient.rpc('verify_admin_password', { input_password: password });
                if (data === true) {
                    closeAdminModal();
                    document.getElementById('login-section').classList.add('hidden');
                    document.getElementById('admin-dashboard-section').classList.remove('hidden');
                    loadAdminData();
                } else { alert("비밀번호가 틀렸습니다."); }
            } catch (e) { alert("관리자 기능을 사용할 수 없습니다. (SQL 설정 확인 필요)"); }
        }

        async function loadAdminData() {
            const tbody = document.getElementById('admin-table-body');
            tbody.innerHTML = '<tr><td colspan="7" class="text-center py-4">로딩 중...</td></tr>';
            const { data: students } = await supabaseClient.from('student_data').select('*').order('id');
            document.getElementById('admin-student-count').textContent = `학생: ${students.length}명`;
            tbody.innerHTML = '';
            students.forEach(s => {
                const check = (d) => d && Object.keys(d).length > 0 ? '✅' : '❌';
                const tr = document.createElement('tr');
                tr.innerHTML = `<td class="py-3 pl-2">${s.id}</td><td>${s.name}</td>
                    <td class="text-center">${check(s.s1)}</td><td class="text-center">${check(s.s2)}</td>
                    <td class="text-center">${check(s.s3)}</td><td class="text-center">${check(s.s4)}</td>
                    <td class="text-right pr-2"><button onclick='downloadHtml(${JSON.stringify(s)})' class="bg-blue-50 text-blue-600 px-2 py-1 rounded text-xs font-bold">다운로드</button></td>`;
                tbody.appendChild(tr);
            });
        }

        function downloadHtml(student) {

                <body><div class="card"><h1>📊 통계 포스터 보고서</h1><p>학번: ${student.id} | 성명: ${student.name}</p><hr>
                <h3>1차시:</h3><pre>${JSON.stringify(student.s1, null, 2)}</pre>
                <h3>2~3차시:</h3><pre>${JSON.stringify(student.s2, null, 2)}</pre>
                <h3>4~5차시:</h3><pre>${JSON.stringify(student.s3, null, 2)}</pre>
                <h3>6차시:</h3><pre>${JSON.stringify(student.s4, null, 2)}</pre></div></body></html>`;
            const blob = new Blob([content], { type: 'text/html' });
            const a = document.createElement('a');
            a.href = URL.createObjectURL(blob);
            a.download = `보고서_${student.id}_${student.name}.html`;
            a.click();
        }
